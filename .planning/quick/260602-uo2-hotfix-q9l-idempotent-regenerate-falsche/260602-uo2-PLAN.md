async fn update(
    &self,
    entity: &MemberDocumentEntity,
    _process: &str,
    tx: Self::Transaction,
) -> Result<(), DaoError> {
    let id = entity.id.as_bytes().to_vec();
    let old_version = entity.version.as_bytes().to_vec();        // <-- entity.version IS the old version
    let new_version = Uuid::new_v4().as_bytes().to_vec();        // <-- DAO rotates internally
    // ...
    let rows_affected = sqlx::query(
        "UPDATE member_document SET ..., version = ?, ... \
         WHERE id = ? AND version = ? AND deleted IS NULL",
    )
    .bind(new_version)   // SET version = new_version
    .bind(id)
    .bind(old_version)   // WHERE version = old_version  <-- THIS is why entity.version must be the OLD one
    .execute(...);

    if rows_affected == 0 {
        return Err(DaoError::ConflictError(Arc::from("Version mismatch")));
    }
}
