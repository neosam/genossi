    ```rust
    /// Phase 12 D-18: Parst die URL-Query-Params des Repayment-Mail-Redirects.
    /// `search` ist der raw search-String inkl. fuehrendem `?` (oder ohne;
    /// UrlSearchParams::new_with_str akzeptiert beides).
    pub fn parse_mail_query(search: &str) -> ParsedMailContext {
        // Manuelles String-Splitting fuer Testbarkeit ohne web_sys-Dependency
        // (Tests laufen in nativen cargo-test-Targets, kein WASM).
        let mut phase_id: Option<Uuid> = None;
        let mut member_ids: Vec<Uuid> = Vec::new();

        let s = search.trim_start_matches('?');
        for pair in s.split('&') {
            let mut kv = pair.splitn(2, '=');
            let key = kv.next().unwrap_or("");
            let value = kv.next().unwrap_or("");
            match key {
                "phase_id" => {
                    if let Ok(u) = Uuid::parse_str(value) {
                        phase_id = Some(u);
                    }
                }
                "members" => {
                    member_ids = value.split(',').filter_map(|s| Uuid::parse_str(s.trim()).ok()).collect();
                }
                _ => {} // ignore `from=repayment` und andere
            }
        }
        ParsedMailContext { phase_id, member_ids }
    }

    pub struct ParsedMailContext {
        pub phase_id: Option<Uuid>,
        pub member_ids: Vec<Uuid>,
    }
    ```
