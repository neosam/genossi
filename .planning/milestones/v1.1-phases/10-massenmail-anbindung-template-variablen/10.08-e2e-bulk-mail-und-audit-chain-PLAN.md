```rust
/// Phase 10 (Plan 08 Task 0): E2E-setup that ALSO spawns the mail worker.
/// Uses an in-memory SqlitePool (matches existing setup()), seeds SMTP config with
/// values that guarantee deterministic failure: host="127.0.0.1", port=1 (no listener),
/// tls="none", user=pass="". The worker will:
///   - lokal RFC5321-parse to_address before SMTP-Connect — invalid addresses fail
///     instantly with AddressError ("not-an-email" -> fail-fast).
///   - For valid addresses, SMTP-Connect to 127.0.0.1:1 fails (Connection refused);
///     worker.send_result is Err, status="failed", description contains "[FAILED:".
///
/// Both code paths produce MemberDocument rows -> SC#4 "kein All-or-Nothing" verified,
/// even without a real Mock-SMTP transport. Differences in error-Subtype (AddressError
/// vs. ConnectionError) are asserted in test E2.
async fn setup_with_mail_worker() -> (
    genossi_rest::test_server::test_support::TestServer,
    Arc<SqlitePool>,
) {
    let pool = Arc::new(
        SqlitePool::connect("sqlite::memory:")
            .await
            .expect("Failed to create in-memory database"),
    );
    sqlx::migrate!("../migrations/sqlite")
        .run(&*pool)
        .await
        .expect("Failed to run migrations");

    // Seed SMTP config so the worker can load it without panicking.
    // SmtpConfig fields are read by load_smtp_config(); use config_service rows.
    // host=127.0.0.1, port=1 -> connect refused; from=test@example.com to satisfy parse.
    // mail_send_interval_seconds=0 so the worker polls fast in tests.
    seed_mail_test_config(&pool).await;

    let rest_state = RestStateImpl::new(pool.clone());
    rest_state.start_mail_worker();
    let server =
        genossi_rest::test_server::test_support::start_test_server(rest_state).await;
    (server, pool)
}

/// Insert minimal config rows so genossi_mail::service::load_smtp_config returns
/// values that drive the worker through the local-RFC5321-parse path. Implementation
/// uses raw sqlx::query on the in-memory pool (avoids needing a ConfigServiceImpl).
async fn seed_mail_test_config(pool: &SqlitePool) {
    // Keys match the names used by load_smtp_config() and get_send_interval().
    // Update these strings if genossi_mail/src/service.rs renames the keys.
    let entries = [

        ("smtp_tls", "none"),
        ("smtp_user", ""),

        ("smtp_from", "test-bulk@example.com"),
        ("mail_send_interval_seconds", "0"),

    for (k, v) in entries {

    }
}
```

**Schritt 2 — wait_for_mail_worker_idle Helper:** Polling auf den Mail-Job-Status. Da im Worker mehrere Jobs gleichzeitig laufen koennen, polle den Status des juengsten Jobs:

```rust
/// Phase 10: Poll until the mail worker has processed all recipients of a job
/// (status transitions away from "running"). Bounded by timeout to avoid hang.
async fn wait_for_mail_worker_idle(
    server: &genossi_rest::test_server::test_support::TestServer,
    job_id: Uuid,
    timeout: std::time::Duration,
) {
    let client = reqwest::Client::new();
    let deadline = std::time::Instant::now() + timeout;
    let url = server.url(&format!("/api/mail/jobs/{}", job_id));
    while std::time::Instant::now() < deadline {
        if let Ok(res) = client.get(&url).send().await {
            if let Ok(body) = res.json::<serde_json::Value>().await {
                let status = body.get("status").and_then(|s| s.as_str()).unwrap_or("");
                if status == "done" || status == "failed" {
                    return;
                }
            }
        }
        tokio::time::sleep(std::time::Duration::from_millis(100)).await;
    }
    panic!(
        "wait_for_mail_worker_idle timed out after {:?} for job {}",
        timeout, job_id
    );
}
```

```rust
struct TestMemberDocumentRow {
    id: Uuid,
    member_id: Uuid,
    document_type: String,

    mail_recipient_id: Option<Uuid>,

/// Phase 10: Query all MemberDocuments produced by the mail worker for a given
/// document_type (typically "repayment_mail"). Uses raw sqlx since this is the
/// DAO layer's concern and we want to avoid adding a test-only REST route.
async fn query_documents_by_type(
    pool: &SqlitePool,
    document_type: &str,
) -> Vec<TestMemberDocumentRow> {
    let rows = sqlx::query(
        "SELECT id, member_id, document_type, description, template_id, mail_recipient_id, status
         FROM member_document
         WHERE document_type = ?
           AND deleted IS NULL
         ORDER BY created ASC",
    )
    .bind(document_type)
    .fetch_all(pool)
    .await
    .expect("query_documents_by_type failed");

    rows.into_iter()
        .map(|r| TestMemberDocumentRow {
            id: uuid_from_blob(&r, "id"),
            member_id: uuid_from_blob(&r, "member_id"),
            document_type: r
                .try_get::<String, _>("document_type")
                .unwrap_or_default(),
            description: r.try_get::<Option<String>, _>("description").ok().flatten(),
            template_id: uuid_from_blob_opt(&r, "template_id"),
            mail_recipient_id: uuid_from_blob_opt(&r, "mail_recipient_id"),
            status: r.try_get::<Option<String>, _>("status").ok().flatten(),
        })
        .collect()
}

/// Helper: convert a BLOB column into Uuid (SQLite stores UUIDs as BLOB per project
/// conventions in genossi_dao_impl_sqlite).
fn uuid_from_blob(row: &sqlx::sqlite::SqliteRow, col: &str) -> Uuid {
    use sqlx::Row;
    let bytes: Vec<u8> = row.try_get(col).unwrap_or_default();
    Uuid::from_slice(&bytes).unwrap_or_else(|_| Uuid::nil())
}
fn uuid_from_blob_opt(row: &sqlx::sqlite::SqliteRow, col: &str) -> Option<Uuid> {
    use sqlx::Row;
    let bytes: Option<Vec<u8>> = row.try_get(col).ok();
    bytes.and_then(|b| Uuid::from_slice(&b).ok())
}
```

```rust
#[tokio::test]
async fn test_bulk_repayment_mail_failure_does_not_block_others() {
    let (server, pool) = setup_with_mail_worker().await;
    let client = reqwest::Client::new();
    let phase_id = create_repayment_phase(&server, 2026, 2000).await;
    let m1 = create_member_with_exit_date(&server, 1, "Max", "Mueller", "m1@example.com", 2026).await;
    let m2 = create_member_with_exit_date(&server, 2, "Anna", "Schmidt", "m2@example.com", 2026).await;
    let m3 = create_member_with_exit_date(&server, 3, "Otto", "Lange", "m3@example.com", 2026).await;
    open_repayment_phase(&server, phase_id).await;
    let tpl_id = create_mail_template(&server, "Subj", "Body {{ first_name }}").await;

    let res = client.post(server.url("/api/mail/send-bulk"))
        .json(&serde_json::json!({
            "to_addresses": [
                {"member_id": m1.to_string(), "to_address": "m1@example.com"},
                {"member_id": m2.to_string(), "to_address": "not-an-email"},  // <-- RFC5321 invalid
                {"member_id": m3.to_string(), "to_address": "m3@example.com"},
            ],
            "subject": "Subj",
            "body": "Body {{ first_name }}",
            "template_id": tpl_id.to_string(),
            "repayment_phase_id": phase_id.to_string(),
        }))
        .send().await.unwrap();
    assert_eq!(res.status(), 200);

    let body: serde_json::Value = res.json().await.unwrap();
    let job_id = Uuid::parse_str(body["id"].as_str().unwrap()).unwrap();
    wait_for_mail_worker_idle(&server, job_id, std::time::Duration::from_secs(20)).await;

    let docs = query_documents_by_type(&pool, "repayment_mail").await;
    assert_eq!(docs.len(), 3,
        "SC#4: All 3 recipients must produce a MemberDocument (kein All-or-Nothing)");

    // Differentiate error pathways: address-parse fail vs. SMTP-connect fail.
    // In stub-SMTP mode (host=127.0.0.1, port=1) both pathways yield status='failed',
    // BUT the description text differs:
    //   - "not-an-email" -> lettre::address::AddressError -> "Invalid to address" in error fmt
    //   - "m1@example.com" -> tcp connect refused -> "Connection refused" or similar TCP error
    let address_fails = docs.iter()
        .filter(|d| d.description.as_deref()
            .map(|s| s.to_lowercase().contains("invalid to address") || s.contains("address"))
            .unwrap_or(false))
        .count();
    let connection_fails = docs.iter()
        .filter(|d| d.description.as_deref()
            .map(|s| s.to_lowercase().contains("connection") || s.contains("refused") || s.contains("connect"))
            .unwrap_or(false))
        .count();
    assert!(address_fails >= 1,
        "SC#4: At least 1 recipient must fail via RFC5321 AddressError ('not-an-email'); got descriptions: {:?}",
        docs.iter().map(|d| d.description.clone()).collect::<Vec<_>>());
    assert!(connection_fails >= 1,
        "SC#4: At least 1 recipient must fail via SMTP connection refused (valid syntax + stub SMTP host); got descriptions: {:?}",
        docs.iter().map(|d| d.description.clone()).collect::<Vec<_>>());

    // Universal: all failed docs contain "[FAILED:" suffix
    for d in &docs {
        let desc = d.description.as_deref().unwrap_or("");
        assert!(desc.contains("[FAILED:"),
            "Each failed doc must carry the [FAILED: suffix, got: {}", desc);
    }
}
```
