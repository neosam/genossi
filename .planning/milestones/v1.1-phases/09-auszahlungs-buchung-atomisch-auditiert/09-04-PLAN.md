```rust
/// Phase 9 PAYO-01 + PAYO-02 — SC #1 + SC #3:
/// Atomarer Auszahlungs-Cascade plus Audit-Chain-Konsistenz.
///
/// Setup: Open-Phase fiscal_year=2026, Member exit_date=2026, current_shares=10,
///        Auto-Entry mit share_count_to_pay_out=10 wird automatisch angelegt.
///        Plan 09-04 setzt das auf share_count_to_pay_out=3 fuer den Test runter
///        per PUT (Phase-8-edit-Pfad), damit Cascade etwas zu tun hat.
///
/// Cascade: POST /mark-paid-out → 200 + RepaymentEntryTO mit status=PaidOut.
/// Verify:
///  - GET Entry → status=PaidOut, neue version.
///  - GET Member → current_shares=7, action_count erhoeht um 1.
///  - GET /api/audit/verify → valid=true (Hash-Chain stabil).
///  - GET /api/audit/member/{member_id} → enthaelt Eintraege mit
///    process="repayment-entry.mark-paid-out" und field_name in
///    {"current_shares", "action_count"}.
///  - GET /api/audit/repayment_entry/{entry_id} → letzter Status-Change
///    hat process="repayment-entry.mark-paid-out" + new_value="PaidOut".
#[tokio::test]
async fn test_mark_paid_out_happy_path_cascade() {
    let server = setup().await;
    let client = reqwest::Client::new();

    // Setup: Open-Phase + Member mit exit_date im fiscal_year
    let fiscal_year = 2026i32;
    let phase_id = create_open_repayment_phase(&client, &server, fiscal_year, 20000).await; // share_value 200,00 EUR
    let member_id = create_member_with_exit_date(&client, &server, fiscal_year, 10).await;

    // Auto-Fill hat einen Entry erzeugt (share_count_to_pay_out=10); finde ihn:
    let entries_resp = client
        .get(server.url(&format!("/api/repayment-entry?phase_id={}", phase_id)))
        .send()
        .await
        .unwrap();
    assert_eq!(entries_resp.status(), reqwest::StatusCode::OK);
    let entries: Vec<serde_json::Value> = entries_resp.json().await.unwrap();
    let entry_v = entries
        .iter()
        .find(|e| e["member_id"].as_str() == Some(&member_id.to_string()))
        .expect("Auto-Fill must have created an entry for our test member");
    let entry_id: Uuid = entry_v["id"].as_str().unwrap().parse().unwrap();
    let entry_version_pre: Uuid = entry_v["version"].as_str().unwrap().parse().unwrap();

    // Get Member.action_count VOR der Cascade als Baseline:
    let member_pre_resp = client.get(server.url(&format!("/api/member/{}", member_id))).send().await.unwrap();
    let member_pre: serde_json::Value = member_pre_resp.json().await.unwrap();
    let current_shares_pre = member_pre["current_shares"].as_i64().unwrap() as i32;
    let action_count_pre = member_pre["action_count"].as_i64().unwrap() as i32;
    assert_eq!(current_shares_pre, 10, "setup invariant: current_shares=10");

    // Optional: Entry auf share_count_to_pay_out=3 anpassen (statt Auto-Fill-Wert 10),
    // damit nach der Cascade current_shares=7 ist statt 0. Falls Auto-Fill-Default
    // schon =10 ist und das fuer den Test OK ist, dieser Schritt entfaellt; sonst
    // PUT mit edit-Body (Phase-8-Pattern):
    let edit_body = serde_json::json!({
        "share_count_to_pay_out": 3,
        "version": entry_version_pre,
    });
    let edit_resp = client
        .put(server.url(&format!("/api/repayment-entry/{}", entry_id)))
        .json(&edit_body)
        .send()
        .await
        .unwrap();
    assert_eq!(edit_resp.status(), reqwest::StatusCode::OK, "PUT edit must succeed; body: {}", edit_resp.text().await.unwrap_or_default());

    // ===== Cascade-Trigger =====
    let mark_url = server.url(&format!("/api/repayment-entry/{}/mark-paid-out", entry_id));
    let mark_resp = client.post(&mark_url).send().await.unwrap();
    assert_eq!(mark_resp.status(), reqwest::StatusCode::OK,
        "PAYO-01: mark_paid_out must succeed; body: {}",
        mark_resp.text().await.unwrap_or_default());
    let entry_to: serde_json::Value = client
        .get(server.url(&format!("/api/repayment-entry/{}", entry_id)))
        .send().await.unwrap()
        .json().await.unwrap();
    assert_eq!(entry_to["status"].as_str(), Some("PaidOut"), "Entry must be PaidOut after Cascade");

    // ===== Verify: Member-Effekte =====
    let member_post_resp = client.get(server.url(&format!("/api/member/{}", member_id))).send().await.unwrap();
    let member_post: serde_json::Value = member_post_resp.json().await.unwrap();
    let current_shares_post = member_post["current_shares"].as_i64().unwrap() as i32;
    let action_count_post = member_post["action_count"].as_i64().unwrap() as i32;
    assert_eq!(current_shares_post, current_shares_pre - 3,
        "PAYO-02: current_shares must drop by share_count_to_pay_out (10 - 3 = 7)");
    assert_eq!(action_count_post, action_count_pre + 1,
        "PAYO-01: action_count must increment by 1 for the new MemberAction::Verkauf");

    // ===== Verify: MemberAction::Verkauf existiert =====
    // (REST-Pfad fuer member_action koennte je nach Phase-1/2-Routing-Konvention
    // anders sein — pragmatisch: GET /api/member/{id}/actions oder
    // GET /api/member-action?member_id=... . Wenn unklar: skip MemberAction-Direct-Check
    // und verlasse dich auf Audit-Trail-Assertion unten.)
    // -- optionaler Block:
    let actions_resp = client
        .get(server.url(&format!("/api/member-action?member_id={}", member_id)))
        .send().await;
    if let Ok(r) = actions_resp {
        if r.status() == reqwest::StatusCode::OK {
            let actions: Vec<serde_json::Value> = r.json().await.unwrap_or_default();
            let verkauf = actions.iter().find(|a| a["action_type"].as_str() == Some("Verkauf"));
            if let Some(v) = verkauf {
                assert_eq!(v["shares_change"].as_i64(), Some(-3),
                    "MemberAction::Verkauf must have shares_change=-3");
                let comment = v["comment"].as_str().unwrap_or("");
                assert!(comment.starts_with("Anteils-Rueckzahlung Phase "),
                    "MemberAction.comment must follow auto-format; got: {}", comment);
            }
        }
    }

    // ===== Verify: Audit-Chain valide (SC #3) =====
    let verify_resp = client.get(server.url("/api/audit/verify")).send().await.unwrap();
    assert_eq!(verify_resp.status(), reqwest::StatusCode::OK);
    let verify_body: serde_json::Value = verify_resp.json().await.unwrap();
    assert_eq!(verify_body["valid"].as_bool(), Some(true),
        "SC #3: Audit-Hash-Chain must remain valid after mark_paid_out Cascade; broken_links: {:?}",
        verify_body["broken_links"]);

    // ===== Verify: Audit-Eintraege mit process="repayment-entry.mark-paid-out" =====
    let member_audit_resp = client
        .get(server.url(&format!("/api/audit/member/{}", member_id)))
        .send().await.unwrap();
    let member_audit: Vec<serde_json::Value> = member_audit_resp.json().await.unwrap();
    let cascade_member_entries: Vec<_> = member_audit
        .iter()
        .filter(|e| e["process"].as_str() == Some("repayment-entry.mark-paid-out"))
        .collect();
    assert!(!cascade_member_entries.is_empty(),
        "SC #3: Member-Audit must have entries with process='repayment-entry.mark-paid-out'");
    let field_names: std::collections::HashSet<&str> = cascade_member_entries
        .iter()
        .map(|e| e["field_name"].as_str().unwrap_or(""))
        .collect();
    assert!(field_names.contains("current_shares"),
        "Member-Audit must contain field_name='current_shares'; got: {:?}", field_names);
    assert!(field_names.contains("action_count"),
        "Member-Audit must contain field_name='action_count'; got: {:?}", field_names);

    let entry_audit_resp = client
        .get(server.url(&format!("/api/audit/repayment_entry/{}", entry_id)))
        .send().await.unwrap();
    let entry_audit: Vec<serde_json::Value> = entry_audit_resp.json().await.unwrap();
    let status_change = entry_audit
        .iter()
        .rev()
        .find(|e| e["field_name"].as_str() == Some("status"))
        .expect("Entry-Audit must contain a status field-change entry");
    assert_eq!(status_change["process"].as_str(), Some("repayment-entry.mark-paid-out"));
    assert_eq!(status_change["new_value"].as_str(), Some("PaidOut"));
}
```

**Test 2 — `test_mark_paid_out_validates_insufficient_shares` (PAYO-03, SC #2):**

```rust
/// Phase 9 PAYO-03 / SC #2: Validation blockt mark_paid_out wenn
/// Member.current_shares < entry.share_count_to_pay_out.
///
/// Setup: Member current_shares=2; manueller Entry mit share_count_to_pay_out=5.
/// (NB: Service-Pre-Check fuer manuelle Entries verlangt share_count <= current_shares,
/// daher MUSS hier ein Trick angewendet werden: erst Member mit current_shares=10
/// + Entry share_count=5, dann den Member auf current_shares=2 reduzieren via
/// MemberAction::Verkauf-Direkt-Insert ODER per zweitem mark_paid_out auf einem
/// anderen Entry. Pragmatischste Methode: setze den Entry an einem Member mit
/// shares_at_joining=5, der durch eine bestehende Verkauf-Action bereits
/// current_shares=2 hat.)
///
/// Erwartung: POST mark-paid-out → 400; body.contains("share_count_to_pay_out").
#[tokio::test]
async fn test_mark_paid_out_validates_insufficient_shares() {
    let server = setup().await;
    let client = reqwest::Client::new();

    let fiscal_year = 2026i32;
    let phase_id = create_open_repayment_phase(&client, &server, fiscal_year, 20000).await;
    // Member mit current_shares=5 (shares_at_joining=5), exit_date im fiscal_year
    let member_id = create_member_with_exit_date(&client, &server, fiscal_year, 5).await;

    // Auto-Fill erzeugt Entry mit share_count_to_pay_out=5.
    let entries: Vec<serde_json::Value> = client
        .get(server.url(&format!("/api/repayment-entry?phase_id={}", phase_id)))
        .send().await.unwrap()
        .json().await.unwrap();
    let entry_v = entries
        .iter()
        .find(|e| e["member_id"].as_str() == Some(&member_id.to_string()))
        .expect("Auto-Fill must have created an entry");
    let entry_id: Uuid = entry_v["id"].as_str().unwrap().parse().unwrap();

    // Schritt 1: Reduziere Member.current_shares per Direkt-MemberAction (Verkauf):
    let action_body = serde_json::json!({
        "member_id": member_id,
        "action_type": "Verkauf",
        "date": format!("{}-01-01", fiscal_year),
        "shares_change": -3,
        "comment": "Phase-9-test-setup: drop current_shares to 2"
    });
    let action_resp = client
        .post(server.url("/api/member-action"))
        .json(&action_body)
        .send().await.unwrap();
    assert!(action_resp.status().is_success(),
        "Setup-Verkauf must succeed to drop current_shares; status={}, body={}",
        action_resp.status(), action_resp.text().await.unwrap_or_default());

    // Verify: Member.current_shares ist jetzt 2.
    let member_check: serde_json::Value = client.get(server.url(&format!("/api/member/{}", member_id)))
        .send().await.unwrap()
        .json().await.unwrap();
    assert_eq!(member_check["current_shares"].as_i64(), Some(2),
        "setup invariant: current_shares=2 after manual Verkauf-Setup");

    // ===== Cascade-Trigger: PAYO-03 muss feuern =====
    let mark_url = server.url(&format!("/api/repayment-entry/{}/mark-paid-out", entry_id));
    let mark_resp = client.post(&mark_url).send().await.unwrap();
    assert_eq!(mark_resp.status(), reqwest::StatusCode::BAD_REQUEST,
        "PAYO-03: mark_paid_out must reject with 400 when current_shares < share_count_to_pay_out");
    let body = mark_resp.text().await.unwrap_or_default();
    assert!(body.contains("share_count_to_pay_out"),
        "PAYO-03 error body must reference field 'share_count_to_pay_out'; body: {}", body);
    assert!(body.contains("2") && body.contains("5"),
        "PAYO-03 error body must include both values (current=2, requested=5); body: {}", body);

    // Verify-Tail: Entry ist UNVERAENDERT (Status=Open, kein PaidOut), Audit-Chain valide.
```

**Test 4 — `test_mark_paid_out_race_one_succeeds_one_conflicts` (D-12 / SC #5):**

Wortgleich aus PATTERNS.md §"Test-Funktions-Skelett (Phase 9)" — angepasst auf vorhandene Setup-Helper. **D-12 erfordert EXAKT** tokio::join! als Parallelisierungs-Mechanismus + sorted-Statuses-Assertion [200, 409]:

```rust
/// Phase 9 SC #5 / D-12: Race-Defense via tokio::join!.
/// Zwei parallele mark_paid_out-Aufrufe auf demselben Entry → genau ein 200
/// (Cascade durchgegangen) und ein 409 (Version-Mismatch im DAO-Update
/// `WHERE version = ?` — siehe genossi_dao_impl_sqlite/src/repayment_entry.rs:160-180).
/// Pattern aus Phase 2 HLPR-04 (e2e_tests.rs:8783-8821), Status-Code-Wechsel
/// von 410 (Helper-Token-already-used) auf 409 (Version-Mismatch).
/// D-12: NIE [200, 200] (waere Double-Verkauf) und NIE [409, 409] (waere Total-Deadlock).
#[tokio::test]
async fn test_mark_paid_out_race_one_succeeds_one_conflicts() {
    let server = setup().await;
    let client = reqwest::Client::new();

    let fiscal_year = 2026i32;
    let phase_id = create_open_repayment_phase(&client, &server, fiscal_year, 20000).await;
    let member_id = create_member_with_exit_date(&client, &server, fiscal_year, 5).await;

    let entries: Vec<serde_json::Value> = client
        .get(server.url(&format!("/api/repayment-entry?phase_id={}", phase_id)))
        .send().await.unwrap()
        .json().await.unwrap();
    let entry_id: Uuid = entries
        .iter()
        .find(|e| e["member_id"].as_str() == Some(&member_id.to_string()))
        .map(|e| e["id"].as_str().unwrap().parse().unwrap())
        .expect("Auto-Fill must have created an entry");

    let url = server.url(&format!("/api/repayment-entry/{}/mark-paid-out", entry_id));

    // D-12: Beide POSTs parallel via tokio::join! (KEIN sequenzieller await):
    let (resp_a, resp_b) = tokio::join!(
        client.post(&url).send(),
        client.post(&url).send(),
    );
    let status_a = resp_a.unwrap().status();
    let status_b = resp_b.unwrap().status();

    let mut statuses = [status_a, status_b];
    statuses.sort_by_key(|s| s.as_u16());
    assert_eq!(statuses[0], reqwest::StatusCode::OK,
        "SC #5 / D-12: exactly one race-call must succeed (200); got {:?}", statuses);
    assert_eq!(statuses[1], reqwest::StatusCode::CONFLICT,
        "SC #5 / D-12: exactly one race-call must conflict (409 via version-mismatch); got {:?}", statuses);

    // Defense-in-Depth: final state of entry is PaidOut (winner committed).
    let entry_post: serde_json::Value = client.get(server.url(&format!("/api/repayment-entry/{}", entry_id)))
        .send().await.unwrap()
        .json().await.unwrap();
    assert_eq!(entry_post["status"].as_str(), Some("PaidOut"),
        "Winner's commit must persist; entry must be PaidOut after race");
}
```

**WICHTIG — fragile Stellen:**

1. **REST-Pfade verifizieren:** Wenn beispielsweise `GET /api/repayment-entry?phase_id=X` in der Codebase eigentlich `GET /api/repayment-entry/by-phase/X` heisst (o.ae.), den TATSAECHLICHEN Pfad aus den Phase-7/8-Tests grep'en und korrigieren.

2. **Audit-Endpoint-Pfade:** `/api/audit/member/{id}`, `/api/audit/member_action/{id}`, `/api/audit/repayment_entry/{id}` — `entity_type` ist Underscore (`member_action`, `repayment_entry`), siehe RESEARCH Z. 410.

3. **member-action-POST-Pfad in Test 2:** Pfad `POST /api/member-action` muss existieren; wenn nicht, alternativ via Phase-1-Member-Action-Service-Pfad (`/api/member/{id}/actions` o.ae.). Bei Unklarheit: `grep -E "post\(create_member_action" genossi_rest/src/`.

4. **Race-Test Flakiness:** Wenn der Race-Test in CI gelegentlich `[200, 500]` statt `[200, 409]` produziert, ist das ein Hinweis auf zu kurzen sqlx busy_timeout (siehe RESEARCH Pitfall #11). NICHT als `#[ignore]` markieren; stattdessen einen schlichten `tokio::time::sleep(Duration::from_millis(1)).await` vor `tokio::join!` einbauen.

5. **share_count_to_pay_out vs current_shares im Auto-Fill (Test 1):** Auto-Fill setzt `share_count_to_pay_out = Member.current_shares` (ENTR-01). Wenn Member current_shares=10 und Auto-Fill den Entry mit =10 angelegt hat, dann reduziert die Cascade current_shares auf 0 — nicht 7. Wenn das fuer Tests problematisch ist, edit den Entry vor dem mark-paid-out auf =3 (Plan im Test 1 macht das bereits).

6. **action_count vor/nach:** Wenn `create_member_with_exit_date` selbst Member-Actions erzeugt (Austritt-Action), ist `action_count` initial NICHT 0 sondern z.B. 1. Test 1 nutzt **diff-basierte Assertion** (`action_count_post == action_count_pre + 1`) — robust gegen Setup-Aenderungen.
