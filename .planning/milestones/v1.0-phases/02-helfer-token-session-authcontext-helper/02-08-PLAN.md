    async fn test_helper_token_create_returns_qr_and_code() {
        let server = setup().await;
        let client = reqwest::Client::new();
        let assembly_id = create_open_assembly_for_helper_test(&client, &server).await;

        let response = client
            .post(server.url(&format!("/api/assembly/{}/helper-tokens", assembly_id)))
            .json(&serde_json::json!({"memo": "Anna"}))
            .send().await.unwrap();
        assert_eq!(response.status(), StatusCode::CREATED, "create must return 201");
        let body: serde_json::Value = response.json().await.unwrap();

        // HLPR-01 SC: code is 10 chars, qr_svg starts with <
        let code = body["code"].as_str().expect("code field");
        assert_eq!(code.len(), 10, "code must be 10 chars (D-09); got '{}'", code);
        // Crockford alphabet check: each char must be in 0123456789ABCDEFGHJKMNPQRSTVWXYZ
        let alphabet = "0123456789ABCDEFGHJKMNPQRSTVWXYZ";
        for c in code.chars() {
            assert!(alphabet.contains(c), "code contains non-Crockford char '{}': {}", c, code);
        }

        let qr_svg = body["qr_svg"].as_str().expect("qr_svg field");
        assert!(qr_svg.starts_with("<?xml") || qr_svg.starts_with("<svg"),
            "qr_svg must be SVG; got '{}'", &qr_svg[..40.min(qr_svg.len())]);
        assert!(qr_svg.contains("</svg>"), "qr_svg must contain closing tag");

        // Token TO must have status=Open and the memo we set
        assert_eq!(body["token"]["memo"].as_str().unwrap(), "Anna");
        assert_eq!(body["token"]["status"].as_str().unwrap(), "Open");

Sie nutzt den `MockSessionServiceImpl::with_probe(DbAssemblyStatusProbe)`-Aufruf aus Plan 02-07 Step 8 + die Helper-Cookie-Erkennung aus Plan 02-06 Task 2:

    1. Assembly öffnen, Token erzeugen, redeem aufrufen — capture token_id für die Cookie-Konstruktion

Helper-Cookie `helper:<assembly_id>:<token_id>` an einem helper-protected GET senden → MUSS NICHT 401 sein (Cookie ist als Helper-Session ERKANNT; das admin-only Endpoint antwortet mit 403, weil Helper kein admin-Privileg hat — wichtig: 403 bedeutet "Cookie wurde geparsed", nicht "kein Cookie")
    3. close_assembly
    4. Selbes Cookie an dasselbe Endpoint → MUSS 401 sein (D-18: DbAssemblyStatusProbe::is_open(aid) → false → MockSessionServiceImpl::extract_auth_context returnt Ok(None) → auth-middleware antwortet mit 401)

    ```rust
    #[tokio::test]
    async fn test_helper_token_session_invalidated_after_close_assembly() {
        // HLPR-05 (D-18): Eine bestehende Helper-Session (Cookie im Browser) muss nach
        // close_assembly invalidiert sein. Mock-Setup: Plan 02-07 Step 8 verdrahtet
        // DbAssemblyStatusProbe in MockSessionServiceImpl::with_probe — die Probe
        // liest assembly.status aus der DB und liefert false sobald Closed.
        // Plan 02-06 Task 2 lehrt MockSessionServiceImpl, das Cookie-Format
        // "helper:<assembly_uuid>:<token_id>" als Helper-Session zu erkennen.

        let server = setup().await;
        let client = reqwest::Client::new();
        let assembly_id = create_open_assembly_for_helper_test(&client, &server).await;
        let (token_id, code) = create_helper_token_for_test(&client, &server, assembly_id, "Frida").await;

        // (1) Redeem (echter Pfad — der Service erzeugt eine echte Session in der DB).
        let resp = client.post(server.url("/api/helper/redeem"))
            .json(&serde_json::json!({"code": code.clone()}))
            .send().await.unwrap();
        assert_eq!(resp.status(), StatusCode::OK, "redeem must succeed while assembly Open");

        // Helper-format-Cookie für den Mock-Pfad konstruieren (Plan 02-06 Task 2 Format).
        // Das e2e-Test verwendet im mock_auth-Build NICHT den Set-Cookie-UUID aus der
        // redeem-Response, weil MockSessionServiceImpl Helper-Sessions nur über das
        // "helper:<aid>:<tid>"-Format erkennt. Der reale Session-Eintrag in der DB ist
        // hier nicht relevant — wir testen den Cascade-Pfad in extract_auth_context.
        let helper_cookie = format!("app_session=helper:{}:{}", assembly_id, token_id);

        // Pick a helper-protected probe endpoint. We use GET /api/assembly/{aid}/helper-tokens
        // (admin-only). Pass the helper-format cookie:
        //   - Assembly Open  → MockSessionServiceImpl returnt AuthContext::Helper → admin-check fails → 403
        //   - Assembly Closed → DbAssemblyStatusProbe::is_open returnt false → extract_auth_context returnt None → auth-middleware → 401
        let probe_url = server.url(&format!("/api/assembly/{}/helper-tokens", assembly_id));

        // (2) Cookie wird ERKANNT solange Assembly Open ist: MUSS NICHT 401 sein.
        //     Erwartet: 403 (admin-check denies helper). Toleriert wird auch 200/204
        //     falls die Permission-Logik weiter aufweicht — aber NIE 401.
        let resp_open = client.get(&probe_url)
            .header("cookie", helper_cookie.as_str())
            .send().await.unwrap();
        assert_ne!(
            resp_open.status(), StatusCode::UNAUTHORIZED,
            "before close_assembly the helper cookie MUST be recognized (got 401 — Plan 02-06 Task 2 cookie-format support is broken or DbAssemblyStatusProbe is missing)"
        );
        // Sanity: Helper-Sessions dürfen niemals admin-Endpoints durchlassen.
        assert!(
            matches!(resp_open.status(), StatusCode::FORBIDDEN | StatusCode::OK),
            "helper-protected probe must be 403 (admin check) or 200; got {}",
            resp_open.status()
        );

        // (3) close assembly.
        let resp = client.post(server.url(&format!("/api/assembly/{}/close", assembly_id)))
            .send().await.unwrap();
        assert_eq!(resp.status(), StatusCode::OK, "close_assembly must succeed");

        // (4) Selbes Cookie nach close: D-18 cascade muss greifen — 401.
        let resp_closed = client.get(&probe_url)
            .header("cookie", helper_cookie.as_str())
            .send().await.unwrap();
        assert_eq!(
            resp_closed.status(), StatusCode::UNAUTHORIZED,
            "after close_assembly the helper cookie MUST be rejected by D-18 cascade (HLPR-05). \
             Got {} — check that DbAssemblyStatusProbe is wired in genossi_bin (Plan 02-07 Step 8) \
             and that MockSessionServiceImpl invokes the probe (Plan 02-06 Task 2).",
            resp_closed.status()
        );
    }
