       ```rust
       //! Phase 4 Plan 04 — Component-First; reused by /helper/attendance and
       //! /assemblies/{id} Anwesenheits-Tab.
       use dioxus::prelude::*;
       use gloo_timers::future::TimeoutFuture;
       use std::rc::Rc;

       use crate::i18n::{use_i18n, Key};

       const DEBOUNCE_MS: u32 = 500;

       #[component]
       pub fn AttendanceSearch(on_change: EventHandler<String>) -> Element {
           let i18n = use_i18n();
           let mut query = use_signal(String::new);
           let mut debounce_gen = use_signal(|| 0u64); // generation counter to cancel stale fires
           let mut pending = use_signal(|| false);

           rsx! {
               div { class: "relative w-full mb-4",
                   span { class: "absolute left-3 top-1/2 -translate-y-1/2 text-gray-400", "\u{1F50D}" }
                   input {
                       class: "w-full pl-10 pr-10 py-2 border border-gray-300 rounded-md focus:ring-2 focus:ring-blue-500",
                       r#type: "text",
                       placeholder: "{i18n.t(Key::AttendanceSearch)}",
                       value: "{query}",
                       oninput: move |e| {
                           let value = e.value();
                           query.set(value.clone());
                           pending.set(true);
                           let gen = {
                               let mut g = debounce_gen.write();
                               *g += 1;
                               *g
                           };
                           let on_change_evt = on_change;
                           spawn(async move {
                               TimeoutFuture::new(DEBOUNCE_MS).await;
                               // Cancel-stale: only fire if generation hasn't moved forward.
                               if *debounce_gen.read() == gen {
                                   pending.set(false);
                                   on_change_evt.call(value.clone());
                               }
                           });
                       },
                   }
                   if pending() {
                       span { class: "absolute right-3 top-1/2 -translate-y-1/2",
                           span { class: "block h-2 w-2 rounded-full bg-blue-400 animate-pulse" }
                       }

               }
       ```

       ```rust
       //! Live attendance counter for /helper/attendance and /assemblies/{id} Anwesenheits-Tab.
       //! Phase 4 Plan 04 — D-11/D-14 + UI-SPEC §LiveCounter + RESEARCH Pattern 1.
       //!
       //! Polls GET /api/assembly/{id}/stats every 5s; emits ConnState transitions
       //! (Healthy / Warning / Lost) so the parent (HelperShell or assembly_details)
       //! can render the ConnectionBanner.
       //!
       //! Hard constraints:
       //! - Display reads literally "X von Y anwesend" with explicit "von" (ROADMAP SC#3).
       //! - Y always shown when known (Member-Universe-Snapshot is stable for the whole GV).
       //! - X dashed during Lost-state (RESEARCH §Polling-Pattern recovery).
       //! - Stop polling on unmount (Dioxus drops use_future automatically).

       use dioxus::prelude::*;
       use gloo_timers::future::TimeoutFuture;
       use uuid::Uuid;

       use crate::api::{self, AttendanceStatsTO};
       use crate::i18n::{use_i18n, Key};
       use crate::service::config::CONFIG;

       const POLL_INTERVAL_MS: u32 = 5_000;

       #[derive(Copy, Clone, Debug, PartialEq, Eq)]
       pub enum ConnState {
           /// Last poll succeeded.
           Healthy,
           /// One consecutive failure — silent (no banner).
           Warning,
           /// Two or more consecutive failures — banner shown.
           Lost,
       }

       #[component]
       pub fn LiveCounter(
           assembly_id: Uuid,
           polling_enabled: bool,
           on_connection_state: EventHandler<ConnState>,
       ) -> Element {
           let i18n = use_i18n();
           let mut stats = use_signal(|| Option::<AttendanceStatsTO>::None);
           let mut consecutive_failures = use_signal(|| 0u32);

           // Polling loop — auto-cancelled on unmount.
           use_future(move || async move {
               loop {
                   if !polling_enabled {
                       TimeoutFuture::new(1_000).await;
                       continue;
                   }
                   let config = CONFIG.read().clone();
                   match api::get_assembly_stats(&config, assembly_id).await {
                       Ok(s) => {
                           let was_lost = *consecutive_failures.read() >= 2;
                           stats.set(Some(s));
                           consecutive_failures.set(0);
                           if was_lost {
                               on_connection_state.call(ConnState::Healthy);
                           }
                       }
                       Err(_) => {
                           let n = {
                               let mut g = consecutive_failures.write();
                               *g += 1;
                               *g
                           };
                           if n == 1 {
                               on_connection_state.call(ConnState::Warning);
                           } else if n == 2 {
                               // Only emit Lost on the *transition* to 2 failures.
                               on_connection_state.call(ConnState::Lost);
                           }
                       }
                   }
                   TimeoutFuture::new(POLL_INTERVAL_MS).await;
               }
           });

           // Display logic — strictly per UI-SPEC state table.
           let display = match (&*stats.read(), *consecutive_failures.read()) {
               (None, _) => i18n.t(Key::AttendanceCounterUnknown).to_string(),
               (Some(s), 0..=1) => format!("{} von {} anwesend", s.present, s.total),
               (Some(s), _) => format!("— von {} anwesend", s.total),
           };

           rsx! {
               div { class: "bg-white border border-gray-200 rounded-lg p-6 mb-4 flex items-baseline justify-between",
                   span { class: "text-sm font-medium text-gray-500 uppercase tracking-wider",
                       "{i18n.t(Key::AttendanceCounterLabel)}"
                   }
                   span { class: "text-4xl font-bold text-gray-900", "{display}" }
               }
           }
       }
       ```

       ```rust
           on_toggle_success: EventHandler<()>,
       ) -> Element {
           let i18n = use_i18n();
           let mut members = use_signal(|| Vec::<AttendanceMemberTO>::new());
           let mut loading = use_signal(|| true);
           let mut error = use_signal(|| Option::<String>::None);
           // Per-row loading map: member_id -> in-flight
           let mut row_loading = use_signal(|| HashMap::<Uuid, bool>::new());
           // Per-row error: member_id -> message
           let mut row_error = use_signal(|| HashMap::<Uuid, String>::new());

           // Re-fetch on (search_query change | refresh_signal change | mount).
           use_effect(move || {
               let q = search_query();
               let _r = refresh_signal();
               let aid = assembly_id;
               let mut members = members;
               let mut loading = loading;
               let mut error = error;
               let row_loading_snapshot = row_loading();
               spawn(async move {
                   loading.set(true);
                   error.set(None);
                   let config = CONFIG.read().clone();
                   let q_opt = if q.is_empty() { None } else { Some(q.as_str()) };
                   match api::list_attendance_members(&config, aid, q_opt).await {
                       Ok(rows) => {
                           // Pitfall 5: do not overwrite rows currently in `loading`.
                           // Strategy: replace whole list, but keep is_present for in-flight rows
                           // until their request resolves.
                           let merged: Vec<AttendanceMemberTO> = rows.into_iter().map(|mut row| {
                               if *row_loading_snapshot.get(&row.member_id).unwrap_or(&false) {
                                   // Find the existing row and keep its is_present
                                   if let Some(existing) = members.read().iter().find(|m| m.member_id == row.member_id) {
                                       row.is_present = existing.is_present;
                                   }
                               }
                               row
                           }).collect();
                           members.set(merged);
                       }
                       Err(e) => {
                           error.set(Some(e.message));
                       }
                   }
                   loading.set(false);
               });
           });

           // Render
           if *loading.read() && members.read().is_empty() {
               return rsx! { div { class: "flex justify-center py-12", span { "{i18n.t(Key::Loading)}" } } };
           }
           if let Some(msg) = error.read().as_ref() {
               return rsx! {
                   div { class: "bg-red-50 border border-red-200 text-red-700 p-4 rounded-lg",
                       "{msg}"
                   }
               };
           }
           if members.read().is_empty() {
               return rsx! {
                   div { class: "text-center py-12 text-gray-500",
                       p { class: "text-lg font-medium", "{i18n.t(Key::AttendanceEmpty)}" }
                       p { class: "text-sm mt-2", "{i18n.t(Key::AttendanceEmptyHint)}" }
                   }
               };
           }

           rsx! {
               div { class: "flex flex-col gap-2",
                   for m in members.read().iter() {
                       {
                           let mid = m.member_id;
                           let is_present = m.is_present;
                           let row_is_loading = *row_loading.read().get(&mid).unwrap_or(&false);
                           let member_number = m.member_number;
                           let salutation_str = m.salutation.clone().unwrap_or_default();
                           let title_str = m.title.clone().unwrap_or_default();
                           let first_name = m.first_name.clone();
                           let last_name = m.last_name.clone();
                           let row_err = row_error.read().get(&mid).cloned();
                           let aid = assembly_id;

                           // Toggle button class per state-machine (UI-SPEC §Toggle button states)
                           let (btn_class, glyph): (&'static str, &'static str) = if row_is_loading {
                               ("min-h-[44px] min-w-[44px] inline-flex items-center justify-center rounded-md bg-gray-100 border-2 border-gray-300 text-gray-400 cursor-not-allowed", "\u{29D6}") // hourglass-ish placeholder; in real RSX use animate-spin svg
                           } else if is_present {
                               ("min-h-[44px] min-w-[44px] inline-flex items-center justify-center rounded-md bg-green-100 border-2 border-green-500 text-green-800 hover:bg-green-200", "\u{2713}")
                           } else {
                               ("min-h-[44px] min-w-[44px] inline-flex items-center justify-center rounded-md bg-white border-2 border-gray-300 text-gray-700 hover:bg-gray-50", "\u{2610}")
                           };

                           let aria_label = if is_present { i18n.t(Key::AttendanceToggleAbsent).to_string() } else { i18n.t(Key::AttendanceTogglePresent).to_string() };

                           rsx! {
                               div { key: "{mid}", class: "bg-white border border-gray-200 rounded-lg p-3 flex items-center gap-4 hover:bg-gray-50 transition-colors",
                                   div { class: "flex-1 flex flex-col",
                                       div { class: "text-base font-medium text-gray-900",
                                           span { class: "text-gray-500 mr-2", "#{member_number}" }
                                           {if !salutation_str.is_empty() { rsx! { span { class: "mr-1", "{salutation_str}" } } } else { rsx! {} }}
                                           {if !title_str.is_empty() { rsx! { span { class: "mr-1", "{title_str}" } } } else { rsx! {} }}
                                           span { class: "mr-1", "{first_name}" }
                                           span { "{last_name}" }
                                       }
                                       {if let Some(msg) = row_err {
                                           rsx! { p { class: "text-xs text-red-600 mt-1", "{msg}" } }
                                       } else { rsx! {} }}
                                   }
                                   if read_only {
                                       span { class: "text-sm text-gray-500",
                                           if is_present { "{i18n.t(Key::HelperTokenStatusUsed)}" } else { "{i18n.t(Key::AttendanceToggleAbsent)}" }
                                       }
                                   } else {
                                       button {
                                           class: "{btn_class}",
                                           disabled: row_is_loading,
                                           "aria-label": "{aria_label}",
                                           onclick: move |_| {
                                               // Pitfall 5: enter loading state, NO premature is_present flip
                                               row_loading.with_mut(|m| { m.insert(mid, true); });
                                               row_error.with_mut(|m| { m.remove(&mid); });
                                               let mut row_loading = row_loading;
                                               let mut row_error = row_error;
                                               let on_toggle_success = on_toggle_success;
                                               spawn(async move {
                                                   let config = CONFIG.read().clone();
                                                   let result = if is_present {
                                                       api::mark_absent(&config, aid, mid).await
                                                   } else {
                                                       api::mark_present(&config, aid, mid).await
                                                   };
                                                   row_loading.with_mut(|m| { m.insert(mid, false); });
                                                   match result {
                                                       Ok(()) => {
                                                           // SC#6: no Optimistic-UI — bump refresh, parent's poll/refresh re-fetches authoritative state.
                                                           on_toggle_success.call(());
                                                       }
                                                       Err(e) => {
                                                           row_error.with_mut(|m| { m.insert(mid, e.message.clone()); });
                                                       }
                                                   }
                                               });
                                           },
                                           "{glyph}"
                                       }
                                   }
                               }
                           }
                       }
                   }
               }
           }
       }
       ```
