       **Helper für Cookie-Reading** (Top-Level oder im Modul; KEIN Extension<Context>):
       ```rust
       /// Read the `app_session` cookie from the request headers.
       /// Returns None if the cookie is absent or empty.
       /// Pattern matches the SET-side in redeem_helper_token (line 317-319).
       fn read_session_cookie(headers: &http::HeaderMap) -> Option<String> {
           let raw = headers.get(http::header::COOKIE)?.to_str().ok()?;
           raw.split(';')
               .map(|s| s.trim())
               .find_map(|kv| {
                   let mut it = kv.splitn(2, '=');
                   match (it.next(), it.next()) {
                       (Some("app_session"), Some(v)) if !v.is_empty() => Some(v.to_string()),
                       _ => None,
                   }
               })
       }
       ```

       **Handler 1: get_helper_session** (PUBLIC route, KEIN Extension<Context>):
       ```rust
       /// GET /api/helper/session — returns 200 + HelperSessionTO if a valid Helper-Session-Cookie is present.
       /// Used by Frontend (Phase 4 D-06) for Auto-Redirect on /helper-mount.
       #[utoipa::path(
           get,
           path = "/api/helper/session",
           responses(
               (status = 200, description = "Active helper session", body = HelperSessionTO),
               (status = 401, description = "No helper session present"),
           ),
       )]
       #[instrument(skip(state, headers))]
       pub async fn get_helper_session<RestState: RestStateDef + HelperTokenRestState + AssemblyRestState>(
           State(state): State<RestState>,
           headers: http::HeaderMap,
       ) -> Response {
           error_handler((async {
               let session_id_str = read_session_cookie(&headers)
                   .ok_or(RestError::Unauthorized)?;
               // Validate session via SessionService — must be a Helper-Session
               // (i.e. linked to an assembly_id via the Phase-2 HelperTokenService).
               let session_info = state.session_service()
                   .verify_user_session(&session_id_str)
                   .await
                   .map_err(|_| RestError::Unauthorized)?;
               // Look up assembly_id via HelperTokenService (Phase 2 Plan 02-06 created this lookup).
               // If session is NOT a helper-session (e.g. admin OIDC session): returns None → 401.
               let assembly_id = state.helper_token_service()
                   .find_assembly_for_session(&session_id_str)
                   .await
                   .map_err(|_| RestError::Unauthorized)?;
               let assembly = state.assembly_service()
                   .find_by_id(assembly_id)
                   .await
                   .map_err(|_| RestError::Unauthorized)?;
               // expires_at: from session_info.expires_at OR fall back to assembly.closed_at + 24h cap.
               let expires_at = session_info.expires_at_iso(); // helper that returns ISO8601
               let body = HelperSessionTO {
                   assembly_id,
                   assembly_name: assembly.name.to_string(),
                   expires_at,
               };
               let json = serde_json::to_string(&body)
                   .map_err(|_| RestError::InternalError("serialization failed".into()))?;
               Ok(Response::builder()
                   .status(200)
                   .header(http::header::CONTENT_TYPE, "application/json")
                   .body(Body::new(json))
                   .unwrap())
           }).await)
       }
       ```
