    ```rust
        // Setup: 2 Members mit exit_date im fiscal_year, Phase oeffnen.

        assert_eq!(

            .and_then(|h| h.to_str().ok()).unwrap_or_default();

        let bytes = resp.bytes().await.expect("read bytes");
        assert!(bytes.starts_with(b"%PDF-"));
        assert!(bytes.len() > 1000);

        // MemberDocuments verifizieren: 2 unique members → 2 docs.

        assert_eq!(letter_docs_m1.len(), 1, "Member 1 erhaelt 1 MemberDocument (D-13-04 Aggregation ueber 2 Entries)");
    ```

    ```rust
    #[tokio::test]
    async fn test_letter_idempotency_d13_08_and_no_status_toggle_d13_09() {
        let server = test_server::start_with_in_memory_db().await;
        let client = reqwest::Client::new();
        let fiscal_year = 2025;
        let m = create_member_with_exit_date_and_iban(&client, &server, fiscal_year, Some("DE89370400440532013000")).await;
        let phase = create_open_repayment_phase(&client, &server, fiscal_year, 12000).await;
        let entries = list_entries_for_phase(&client, &server, phase.id).await;
        let entry = entries.into_iter().find(|e| e.member_id == m.id).expect("entry");

        // Initial-Status pruefen.
        let status_before = get_entry_status(&client, &server, entry.id).await;
        assert_eq!(status_before, "open");

        // Bulk-Run #1.
        let resp1 = client
            .post(server.url(&format!("/api/repayment-phase/{}/letters/generate", phase.id)))
            .json(&json!({ "entry_ids": [entry.id] }))
            .send().await.expect("POST #1");
        assert_eq!(resp1.status(), StatusCode::OK);

        // Bulk-Run #2 (selber Entry).
        let resp2 = client
            .post(server.url(&format!("/api/repayment-phase/{}/letters/generate", phase.id)))
            .json(&json!({ "entry_ids": [entry.id] }))
            .send().await.expect("POST #2");
        assert_eq!(resp2.status(), StatusCode::OK, "D-13-08: 2. Call = 200 (kein 409)");

        // Beide Runs erzeugten je ein MemberDocument → 2 docs fuer den Member.
        let docs = list_member_documents(&client, &server, m.id).await;
        let letter_docs: Vec<_> = docs.iter().filter(|d| d.document_type == "repayment_letter").collect();
        assert_eq!(letter_docs.len(), 2, "D-13-08: jeder Bulk-Run erzeugt 1 MemberDocument");

        // D-13-09: Backend toucht status NICHT — bleibt 'open'.
        let status_after = get_entry_status(&client, &server, entry.id).await;
        assert_eq!(status_after, "open",
            "D-13-09: Backend MUSS RepaymentEntry.status NICHT auto-togglen");
    }
    ```
