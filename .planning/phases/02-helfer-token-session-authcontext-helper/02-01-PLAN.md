        created TEXT NOT NULL,
        used_at TEXT,

            let format_dt = |dt: &time::PrimitiveDateTime| {

                            "Failed to format datetime for audit field");

        type Transaction: crate::Transaction;

        async fn create(&self, entity: &HelperTokenEntity, process: &str, tx: Self::Transaction) -> Result<(), DaoError>;

        async fn all(&self, tx: Self::Transaction) -> Result<Arc<[HelperTokenEntity]>, DaoError> {
            let all_entities = self.dump_all(tx).await?;
            let active: Vec<HelperTokenEntity> = all_entities
                .iter().filter(|e| e.deleted.is_none()).cloned().collect();
            Ok(active.into())
        }

        async fn find_by_id(&self, id: Uuid, tx: Self::Transaction) -> Result<Option<HelperTokenEntity>, DaoError> {
            let all_entities = self.dump_all(tx).await?;
            Ok(all_entities.iter().find(|e| e.id == id && e.deleted.is_none()).cloned())

            used_at: time::PrimitiveDateTime,
            tx: Self::Transaction,

        /// Returns Some((used_at, revoked_at)) if a row exists for this token_hash

        async fn all_for_assembly(

            assembly_id: Uuid,
            tx: Self::Transaction,

    impl HelperTokenDao for HelperTokenDaoImpl {
        type Transaction = TransactionImpl;

        async fn dump_all(&self, tx: Self::Transaction) -> Result<Arc<[HelperTokenEntity]>, DaoError> {

        async fn create(&self, entity: &HelperTokenEntity, _process: &str, tx: Self::Transaction) -> Result<(), DaoError> {
            // INSERT INTO helper_token (id, assembly_id, memo, token_hash, created, used_at, session_id, revoked_at, deleted, version) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
            let id = entity.id.as_bytes().to_vec();

            sqlx::query(

            )

            .bind(assembly_id)

            .bind(entity.token_hash.to_string())
            .bind(created)
            .bind(used_at)

            .bind(revoked_at)
            .bind(deleted)
            .bind(version)
            .execute(tx.tx.lock().await.as_mut())
            .await
            .map_err(|e| DaoError::DatabaseError(Arc::from(e.to_string())))?;
            Ok(())
        }

        async fn update(&self, entity: &HelperTokenEntity, _process: &str, tx: Self::Transaction) -> Result<(), DaoError> {
            // Pre-condition: row must exist and not soft-deleted (Phase-1-Pattern aus assembly:148-205)
            let id = entity.id.as_bytes().to_vec();
            let exists: i32 = sqlx::query_scalar(
                "SELECT COUNT(*) FROM helper_token WHERE id = ? AND deleted IS NULL"
            ).bind(id.clone()).fetch_one(tx.tx.lock().await.as_mut()).await
            .map_err(|e| DaoError::DatabaseError(Arc::from(e.to_string())))?;
            if exists == 0 { return Err(DaoError::NotFound); }

            let new_version = entity.version.as_bytes().to_vec();
            // Optimistic locking: caller must increment version before calling update.
            // The WHERE-clause checks `version = old_version`. Since the entity already
            // has the new version, we need to bind both old AND new — caller convention
            // (consistent with Phase-1 assembly DAO impl).
            // Simplification: we accept that the caller has updated `entity.version` BEFORE
            // calling update; we read the current DB version, compare, and update with new
            // version. This matches assembly:148-205.
            // Implementation: see assembly.rs:148-205 for exact pattern.
            // Phase-2 helper_token UPDATE statement (SET ALL non-key data fields including
            // session_id/used_at/revoked_at — those mutate during lifecycle):
            let used_at = entity.used_at.as_ref().map(format_dt).transpose()?;
            let revoked_at = entity.revoked_at.as_ref().map(format_dt).transpose()?;
            let deleted = entity.deleted.as_ref().map(format_dt).transpose()?;
            // Read current version for optimistic-lock check. We accept a DB lookup —
            // identical to assembly:148-205.
            let current_version: Vec<u8> = sqlx::query_scalar(
                "SELECT version FROM helper_token WHERE id = ? AND deleted IS NULL"
            ).bind(id.clone()).fetch_one(tx.tx.lock().await.as_mut()).await
            .map_err(|e| DaoError::DatabaseError(Arc::from(e.to_string())))?;
            // Caller passes a NEW version; we treat the old version as the one currently
            // in DB. The caller must guarantee that they read the entity, mutate it, and
            // hand-set a NEW UUID for `version` — Phase-1-pattern.
            //
            // For the UPDATE we use a single statement with the WHERE check.
            // Simpler: rely on the `WHERE version = ?` clause as the optimistic lock.
            // Caller hands us the DESIRED-NEXT entity (with new version); we use `current_version`
            // we just read AS the comparator. If a concurrent updater changed it between
            // our SELECT and our UPDATE, the UPDATE returns 0 rows.
            let result = sqlx::query(
                "UPDATE helper_token SET assembly_id = ?, memo = ?, token_hash = ?, used_at = ?, session_id = ?, revoked_at = ?, deleted = ?, version = ? WHERE id = ? AND version = ? AND deleted IS NULL"
            )
            .bind(entity.assembly_id.as_bytes().to_vec())
            .bind(entity.memo.to_string())
            .bind(entity.token_hash.to_string())
            .bind(used_at)
            .bind(entity.session_id.as_ref().map(|s| s.to_string()))
            .bind(revoked_at)
            .bind(deleted)
            .bind(new_version)
            .bind(id.clone())
            .bind(current_version)
            .execute(tx.tx.lock().await.as_mut())
            .await
            .map_err(|e| DaoError::DatabaseError(Arc::from(e.to_string())))?;

            if result.rows_affected() == 0 {
                return Err(DaoError::ConflictError(Arc::from("Version mismatch")));
            }
            Ok(())
        }

        async fn atomic_redeem(
            &self,
            token_hash: &str,
            used_at: PrimitiveDateTime,
            tx: Self::Transaction,
        ) -> Result<Option<(Uuid, Uuid)>, DaoError> {
            // RESEARCH §Pattern 1 — VERBATIM. Use query_as::<_, RedeemRow> + fetch_optional
            // (NOT query_as! macro — Pitfall 1 SQLx 0.8 RETURNING-Nullability-Bug).
            let used_at_str = format_dt(&used_at)?;
            let row: Option<RedeemRow> = sqlx::query_as::<_, RedeemRow>(
                "UPDATE helper_token \
                 SET used_at = ? \
                 WHERE token_hash = ? \
                   AND used_at IS NULL \
                   AND revoked_at IS NULL \
                   AND deleted IS NULL \
                 RETURNING id, assembly_id"
            )
            .bind(used_at_str)
            .bind(token_hash)
            .fetch_optional(tx.tx.lock().await.as_mut())
            .await
            .map_err(|e| DaoError::DatabaseError(Arc::from(e.to_string())))?;

            Ok(match row {
                Some(r) => Some((Uuid::from_slice(&r.id)?, Uuid::from_slice(&r.assembly_id)?)),
                None => None,
            })
        }

        async fn set_session_id(
            &self,
            token_id: Uuid,
            session_id: &str,
            tx: Self::Transaction,
        ) -> Result<(), DaoError> {
            let id = token_id.as_bytes().to_vec();
            let result = sqlx::query(
                "UPDATE helper_token SET session_id = ? WHERE id = ?"
            )
            .bind(session_id)
            .bind(id)
            .execute(tx.tx.lock().await.as_mut())
            .await
            .map_err(|e| DaoError::DatabaseError(Arc::from(e.to_string())))?;
            if result.rows_affected() == 0 {
                return Err(DaoError::NotFound);
            }
            Ok(())
        }

        async fn lookup_status(
            &self,
            token_hash: &str,
            tx: Self::Transaction,
        ) -> Result<Option<(Option<PrimitiveDateTime>, Option<PrimitiveDateTime>)>, DaoError> {
            let row: Option<LookupStatusRow> = sqlx::query_as::<_, LookupStatusRow>(
                "SELECT used_at, revoked_at FROM helper_token WHERE token_hash = ? AND deleted IS NULL"
            )
            .bind(token_hash)
            .fetch_optional(tx.tx.lock().await.as_mut())
            .await
            .map_err(|e| DaoError::DatabaseError(Arc::from(e.to_string())))?;

            match row {
                None => Ok(None),
                Some(r) => Ok(Some((
                    r.used_at.as_ref().map(|s| parse_datetime(s)).transpose()?,
                    r.revoked_at.as_ref().map(|s| parse_datetime(s)).transpose()?,
                ))),
            }
        }

        async fn all_for_assembly(
            &self,
            assembly_id: Uuid,
            tx: Self::Transaction,

            )
            .bind(aid)
            .fetch_all(tx.tx.lock().await.as_mut())
            .await
            .map_err(|e| DaoError::DatabaseError(Arc::from(e.to_string())))?;
            let entities: Result<Vec<_>, _> = rows.iter().map(HelperTokenEntity::try_from).collect();
            Ok(Arc::from(entities?))
        }
    }
