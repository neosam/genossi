    async fn extract_auth_context(
        &self,
        session_id: Option<String>,
    ) -> Result<Option<AuthContext>, ServiceError> {
        let Some(sid) = session_id else { return Ok(None); };
        let Some(session) = self.verify_user_session(&sid).await? else {
            return Ok(None);
        };

        // Pitfall 2: Early-Return wenn claims.is_none(), um den OIDC-User-Session-Pfad
        // ohne extra DB-Roundtrip zu lassen.
        if let Some(claims_str) = session.claims.as_deref() {
            // Try to parse helper-claims discriminator (D-16).
            if let Ok(parsed) = serde_json::from_str::<HelperClaims>(claims_str) {
                if parsed.kind == "helper" {
                    // D-18 Status-Check: Helper-Session lebt nur, solange assembly.status == Open.
                    let tx = self.transaction_dao.use_transaction(None).await?;
                    let assembly = self.assembly_dao.find_by_id(parsed.assembly_id, tx.clone()).await?;
                    self.transaction_dao.commit(tx).await?;
                    return match assembly {
                        Some(a) if a.status == AssemblyStatus::Open => {
                            Ok(Some(AuthContext::Helper {
                                session_id: session.session_id,
                                assembly_id: parsed.assembly_id,
                            }))
                        }
                        _ => {
                            // Assembly closed/missing → invalidate session and reject.
                            // HLPR-05 SC#4: cookie wird invalidiert, auch wenn Browser noch das alte hat.
                            self.permission_dao.delete_session(&sid).await.ok();
                            Ok(None)
                        }
                    };
                }
            }
            // Fall-through (claims existed but didn't match helper schema): treat as user-session.
            // This preserves backward-compat for any future claim discriminators.
        }

        // Bestehender User-Session-Pfad (Default-Verhalten).
        Ok(Some(AuthContext::Mock(MockContext {
            user_id: session.user_id,
        })))
    }
