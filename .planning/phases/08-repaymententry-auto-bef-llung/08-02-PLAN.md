    `update` mit Pre-Exists-Check + Optimistic-Locking (Phase-7-Lektion Plan 07-02 D-03):
    ```rust
    async fn update(&self, entity: &RepaymentEntryEntity, _process: &str, tx: Self::Transaction) -> Result<(), DaoError> {
        let id_bytes = entity.id.as_bytes().to_vec();

        // Pre-exists check trennt NotFound von ConflictError
        let exists = sqlx::query_scalar::<_, i32>(
            "SELECT COUNT(*) FROM repayment_entry WHERE id = ? AND deleted IS NULL",
        )
        .bind(id_bytes.clone())
        .fetch_one(tx.tx.lock().await.as_mut())
        .await
        .map_err(|e| DaoError::DatabaseError(Arc::from(e.to_string())))?;

        if exists == 0 {
            return Err(DaoError::NotFound);
        }

        // Atomic version-check + UPDATE; rows_affected == 0 → Conflict (Version-Mismatch oder soft-deleted dazwischen)
        let new_version = Uuid::new_v4().as_bytes().to_vec();
        let member_id = entity.member_id.as_bytes().to_vec();
        let phase_id = entity.phase_id.as_bytes().to_vec();
        let deleted = entity.deleted.map(|d| d.assume_utc().format(&time::format_description::well_known::Iso8601::DEFAULT))
            .transpose()
            .map_err(|e| DaoError::DatabaseError(Arc::from(format!("deleted format: {}", e))))?;
        let old_version = entity.version.as_bytes().to_vec();

        let rows_affected = sqlx::query(
            "UPDATE repayment_entry \
             SET member_id = ?, phase_id = ?, share_count_to_pay_out = ?, status = ?, deleted = ?, version = ? \
             WHERE id = ? AND version = ? AND deleted IS NULL",
        )
        .bind(member_id)
        .bind(phase_id)
        .bind(entity.share_count_to_pay_out as i64)
        .bind(entity.status.as_str())
        .bind(deleted)
        .bind(new_version)
        .bind(id_bytes)
        .bind(old_version)
        .execute(tx.tx.lock().await.as_mut())
        .await
        .map_err(|e| DaoError::DatabaseError(Arc::from(e.to_string())))?
        .rows_affected();

        if rows_affected == 0 {
            return Err(DaoError::ConflictError(Arc::from("Version mismatch")));
        }
        Ok(())
    }
    ```

    > Hinweis: Das `update` propagiert die neue version NICHT zurück in die Entity (codebase-weite Konvention — Phase-7-Plan-05-Lektion zu Stale-Retry). Service-Layer ruft danach typischerweise find_by_id auf falls die neue Version benötigt wird, oder akzeptiert den Stale-Retry-Pattern für E2E.
