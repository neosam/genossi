    ```rust
    impl<Deps: RepaymentExportServiceDeps> RepaymentExportServiceImpl<Deps> {
        /// D-10 / D-11 / Pitfall #2: Permission-Funnel-Order:
        /// 1. Load by id (404 if missing)
        /// 2. Admin gate (403); Authentication::Full short-circuits.
        /// 3. Status gate (409): Open ODER Closed akzeptiert; Preparation rejected.
        ///
        /// WICHTIG: Reihenfolge 1 -> 2 -> 3 verhindert Status-Information-Leak an non-admin.
        async fn check_admin_and_phase_status(
            &self,
            phase_id: Uuid,
            context: Authentication<Deps::Context>,
            tx: Deps::Transaction,
        ) -> Result<RepaymentPhaseEntity, ServiceError> {
            // 1. Load (404 if missing).
            let phase = self
                .repayment_phase_dao
                .find_by_id(phase_id, tx)
                .await?
                .ok_or(ServiceError::EntityNotFound(phase_id))?;

            // 2. Admin gate (403). Authentication::Full short-circuits (Phase-6-Pattern, Test-only).
            match &context {
                Authentication::Full => {}
                Authentication::Context(_) => {
                    self.permission_service
                        .check_permission(ADMIN_PRIVILEGE, context)
                        .await?;
                }
            }

            // 3. Status gate (409): D-10 — Open ODER Closed akzeptiert.
            match phase.status {
                RepaymentPhaseStatus::Open | RepaymentPhaseStatus::Closed => {}
                RepaymentPhaseStatus::Preparation => {
                    return Err(ServiceError::Conflict(Arc::from("phase_not_exportable")));
                }
            }

            Ok(phase)
        }
    }
    ```

    ```rust
    #[async_trait]
    impl<Deps: RepaymentExportServiceDeps> RepaymentExportService for RepaymentExportServiceImpl<Deps> {
        type Context = Deps::Context;
        type Transaction = Deps::Transaction;

        async fn export(
            &self,
            phase_id: Uuid,
            format: ExportFormat,
            include: ExportInclude,
            context: Authentication<Self::Context>,
        ) -> Result<RepaymentExport, ServiceError> {
            // Open Tx for Phase + Entries + Members reads.
            let tx = self.transaction_dao.use_transaction(None).await?;

            // Permission-Funnel: returns Phase if admin + Status in {Open, Closed}.
            let phase = self
                .check_admin_and_phase_status(phase_id, context, tx.clone())
                .await?;

            // Read all entries for this phase (DAO filters soft-deleted via Default-Impl).
            let raw_entries: Vec<_> = self
                .repayment_entry_dao
                .find_by_phase_id(phase_id, tx.clone())
                .await?;

            // Read member per entry (N+1 — Discretion choice per RESEARCH §Q5).
            // Build a Vec<(entry, member)> tuple while reading.
            let mut entry_member_pairs: Vec<(
                genossi_dao::repayment_entry::RepaymentEntryEntity,
                genossi_dao::member::MemberEntity,
            )> = Vec::with_capacity(raw_entries.len());
            for entry in raw_entries.into_iter() {
                // D-02: Skip soft-deleted entries (defense-in-depth; find_by_phase_id should already).
                if entry.deleted.is_some() {
                    continue;
                }
                // MemberDao::find_by_id Default-Impl filters soft-deleted (deleted IS NULL).
                let member_opt = self
                    .member_dao
                    .find_by_id(entry.member_id, tx.clone())
                    .await?;
                if let Some(member) = member_opt {
                    entry_member_pairs.push((entry, member));
                }
                // else: member was soft-deleted -> skip (D-02).
            }

            // Pitfall #8: Commit Tx BEFORE PdfGenerator::render_* (sync method, no async tx ops afterwards).
            self.transaction_dao.commit(tx).await?;

            // D-01/D-02: In-memory include-filter.
            entry_member_pairs.retain(|(entry, _)| match include {
                ExportInclude::Open => matches!(
                    entry.status,
                    RepaymentEntryStatus::Open | RepaymentEntryStatus::Contacted
                ),
                ExportInclude::All => true, // alle 3 Stati
                ExportInclude::Paid => matches!(entry.status, RepaymentEntryStatus::PaidOut),
            });

            // D-09: Stable-sort by (member_number ASC, entry.created ASC).
            entry_member_pairs.sort_by(|a, b| {
                a.1.member_number
                    .cmp(&b.1.member_number)
                    .then_with(|| a.0.created.cmp(&b.0.created))
            });

            // D-04 / D-07: Pre-compute amount_str + purpose-String pro Row.
            let enriched_rows: Vec<RepaymentExportRow> = entry_member_pairs
                .iter()
                .map(|(entry, m)| {
                    let amount_cents = (entry.share_count_to_pay_out as i64) * phase.share_value;
                    let amount_str = format!(
                        "{},{:02}",
                        amount_cents / 100,
                        (amount_cents % 100).abs()
                    );
                    // D-04 wortwoertlich; KEIN Komma/Bindestrich; Leerzeichen-getrennt.
                    let purpose = format!(
                        "Anteilsrueckzahlung GJ {} {} {} {}",
                        phase.fiscal_year, m.member_number, m.first_name, m.last_name
                    );
                    RepaymentExportRow {
                        member_number: m.member_number,
                        name: format!("{} {}", m.first_name, m.last_name),
                        // D-06/D-07: leere IBAN -> leerer String (kein Skip, kein Marker).
                        iban: m
                            .bank_account
                            .as_ref()
                            .map(|s| s.to_string())
                            .unwrap_or_default(),
                        share_count: entry.share_count_to_pay_out,
                        amount_str,
                        purpose,
                    }
                })
                .collect();

            // D-18 Phase 6: tracing::info!-Pattern ersetzt Audit-Eintrag.
            tracing::info!(
                target: EXPORT_TARGET,
                phase_id = %phase_id,
                fiscal_year = phase.fiscal_year,
                format = ?format,
                include = ?include,
                rows = enriched_rows.len(),
                "exporting repayment list"
            );

            // Render PDF (sync; tx already committed).
            let bytes = match format {
                ExportFormat::Pdf => self.pdf_generator.render_repayment_list(
                    "auszahlungsliste.typ",
                    &self.template_base,
                    &phase,
                    &enriched_rows,
                )?,
            };

            // D-15 / SC #2: Server-generated filename, kein User-Input.
            let include_str = match include {
                ExportInclude::Open => "open",
                ExportInclude::All => "all",
                ExportInclude::Paid => "paid",
            };
            let filename = format!("auszahlung-{}-{}.pdf", phase.fiscal_year, include_str);

            Ok(RepaymentExport {
                bytes,
                content_type: "application/pdf",
                filename,
            })
        }
    }
    ```
