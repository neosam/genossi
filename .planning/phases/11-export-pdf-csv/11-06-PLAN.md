    /// Verifiziert: 200, application/pdf, %PDF- Magic, Filename "auszahlung-{fy}-open.pdf".
    #[tokio::test]
    async fn test_export_repayment_pdf_open_happy_path() {
        let server = setup_with_templates().await;
        let client = reqwest::Client::new();

        let fiscal_year = 2026;
        // Setup: 2 members with IBAN, 1 open repayment phase.
        let _m1 = create_member_with_exit_date(&client, &server, 101, fiscal_year, 2).await;
        let _m2 = create_member_with_exit_date(&client, &server, 102, fiscal_year, 3).await;

        let resp = client
            .get(server.url(&format!(
                "/api/repayment-phase/{}/export/pdf?include=open",
                phase.id
            )))
            .send()
            .await
            .expect("GET export failed");
        assert_eq!(resp.status(), reqwest::StatusCode::OK);
        assert_eq!(
            resp.headers()
                .get("content-type")
                .and_then(|h| h.to_str().ok())

        );

            .headers()
            .get("content-disposition")

            .to_string();

            cd.contains(&format!("auszahlung-{}-open.pdf", fiscal_year)),
            "Content-Disposition '{}' should contain filename schema",
            cd
        );

        let bytes = resp.bytes().await.expect("read bytes");

    #[tokio::test]
    async fn test_export_repayment_include_filter_changes_payload() {
        let server = setup_with_templates().await;

            .get(server.url(&format!("/api/repayment-entry?phase_id={}", phase.id)))
            .send()
            .await
