```rust
impl From<ServiceError> for RestError {
    fn from(e: ServiceError) -> Self {
        match e {
            ServiceError::EntityNotFound(_) => RestError::NotFound,
            ServiceError::ValidationError(items) => RestError::BadRequest(/* "field: msg, ..." */),
            ServiceError::PermissionDenied | ServiceError::Unauthorized => RestError::Unauthorized,
            ServiceError::Conflict(m) => RestError::Conflict(m.to_string()),
            _ => RestError::InternalError(...),
        }
    }
}
```
