```rust
/// AttendanceService — Helfer/Vorstand-API für die GV-Anwesenheits-Erfassung.
///
/// **Permission funnel** (D-17/D-18): jede Method ruft intern
/// `check_assembly_access(assembly_id, ctx, tx)` als ersten Schritt
/// (verhindert Endpoint-vergisst-Status-Check-Bug). Implementation siehe
/// AttendanceServiceImpl (Plan 05).

/// **No audit** (D-08, ATTN-05): Anwesenheits-Operationen werden bewusst

#[automock(type Context=(); type Transaction = genossi_dao::MockTransaction;)]
#[async_trait]
pub trait AttendanceService {
    type Context: Clone + Debug + PartialEq + Eq + Send + Sync + 'static;
    type Transaction: genossi_dao::Transaction;

    /// ATTN-01 + ATTN-02: reduzierter Member-View mit optionalem Substring-Filter.
    /// Returns: Liste von AttendanceMemberRow (DAO-Projektion mit nur 7 erlaubten Feldern).
    /// Errors: PermissionDenied (Helfer falsche aid / GV nicht Open / nicht-admin),
    ///         EntityNotFound (assembly_id existiert nicht).
    async fn list_members(
        &self,
        assembly_id: Uuid,
        search: Option<String>,
        context: Authentication<Self::Context>,

    /// Errors: PermissionDenied, EntityNotFound (aid oder mid nicht im snapshot — D-27).
    async fn mark_present(
        &self,
        assembly_id: Uuid,
        member_id: Uuid,
        context: Authentication<Self::Context>,
    ) -> Result<(), ServiceError>;

    /// ATTN-04: idempotenter Toggle-Off (UPDATE-Soft-Delete — D-06).
    /// Idempotent auch bei nicht-existierender Row. Snapshot-Check D-27 greift trotzdem.
    /// Errors: PermissionDenied, EntityNotFound (aid oder mid nicht im snapshot).

        &self,

}
```

```rust
/// **Reduced helper-view of a member (D-24, ATTN-01)** — datenschutz-konforme Projektion.
///
/// **PII-Leak-Guard (Pitfall 6 in 03-RESEARCH.md):** Diese Struct hat EXAKT 7 Felder
/// (member_number, first_name, last_name, salutation, title, is_present, member_id).
///
/// **VERBOTEN:** `impl From<&MemberTO> for AttendanceMemberTO` einzufügen — das würde
/// stillschweigend neue MemberTO-Felder (z. B. zukünftige `iban` / `email` /
/// `bank_account`) propagieren und ATTN-01 verletzen. Konversion läuft EXKLUSIV
/// über `From<&genossi_dao::attendance::AttendanceMemberRow>` — ein expliziter
/// 7-Feld-DTO aus dem DAO-Layer mit derselben Whitelist.
///
/// Plan 06 (REST E2E-Tests) verifiziert den Guard mit Whitelist+Blacklist-Iteration
/// auf dem JSON-Response.
#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, ToSchema)]
pub struct AttendanceMemberTO {
    /// Mitgliedsnummer (ATTN-01).
    pub member_number: i64,
    /// Vorname (ATTN-01).
    pub first_name: String,
    /// Nachname (ATTN-01).
    pub last_name: String,
    /// Anrede ("Herr"/"Frau"/"Firma" or null).
    #[serde(skip_serializing_if = "Option::is_none", default)]
    pub salutation: Option<String>,
    /// Akademischer Titel.
    #[serde(skip_serializing_if = "Option::is_none", default)]
    pub title: Option<String>,
    /// Aktueller Anwesenheits-Status (ATTN-03/04).
    pub is_present: bool,
    /// Member-ID — Frontend braucht das für PUT/DELETE-Requests auf
    /// `/api/attendance/{aid}/{mid}`. Kein PII (UUID).
    pub member_id: Uuid,
}

impl From<&genossi_dao::attendance::AttendanceMemberRow> for AttendanceMemberTO {
    fn from(r: &genossi_dao::attendance::AttendanceMemberRow) -> Self {
        Self {
            member_number: r.member_number,
            first_name: r.first_name.to_string(),
            last_name: r.last_name.to_string(),
            salutation: r.salutation.as_deref().map(String::from),
            title: r.title.as_deref().map(String::from),
            is_present: r.is_present,
            member_id: r.member_id,
        }
    }
}

/// Live-Counter (ASSY-04). `{present, total}` für die `X von Y aktiven Mitgliedern`-Anzeige.
#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize, ToSchema)]
pub struct AttendanceStatsTO {
    pub present: u64,
    pub total: u64,
}

impl From<&genossi_service::attendance::AttendanceStats> for AttendanceStatsTO {
    fn from(s: &genossi_service::attendance::AttendanceStats) -> Self {
        Self {
            present: s.present,
            total: s.total,
        }
    }
}
```
