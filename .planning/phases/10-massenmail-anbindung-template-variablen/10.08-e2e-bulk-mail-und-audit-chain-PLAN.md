#[tokio::test]
async fn test_bulk_repayment_mail_creates_member_documents_per_recipient() {
    let test_server = TestServer::new_with_mail_worker().await;  // existing test_server.rs helper
    let admin = test_server.login_as_vorstand().await;

    // Setup: create RepaymentPhase for fiscal_year=2026, share_value=2000 (20 EUR / share)
    let phase_id = create_repayment_phase(&test_server, &admin, 2026, 2000).await;

    // Create 3 members with exit_date in 2026 (uses Phase-8 helper create_member_with_exit_date)
    let m1 = create_member_with_exit_date(&test_server, &admin, "Max", "Mueller", "m1@example.com", 2026).await;
    let m2 = create_member_with_exit_date(&test_server, &admin, "Anna", "Schmidt", "m2@example.com", 2026).await;
    let m3 = create_member_with_exit_date(&test_server, &admin, "Otto", "Lange", "m3@example.com", 2026).await;

    // Open phase -> auto-fills entries; share_count_to_pay_out=current_shares=1 (Phase-8 default)
    open_repayment_phase(&test_server, &admin, phase_id).await;

    // Create a MailTemplate referencing the Phase 10 vars (deutsche Lokalisierung)
    let tpl_id = create_mail_template(
        &test_server, &admin,
        "Auszahlung GJ {{ fiscal_year }}",
        "Hallo {{ first_name }}, dir werden {{ share_count }} Anteile zu insgesamt {{ payout_amount }} EUR ausbezahlt.",
    ).await;

    // POST /api/mail/send-bulk with template_id + repayment_phase_id
    let res = test_server.post("/api/mail/send-bulk", &admin)
        .json(&serde_json::json!({
            "to_addresses": [
                {"member_id": m1.to_string(), "to_address": "m1@example.com"},
                {"member_id": m2.to_string(), "to_address": "m2@example.com"},
                {"member_id": m3.to_string(), "to_address": "m3@example.com"},
            ],
            "subject": "Auszahlung GJ {{ fiscal_year }}",
            "body": "Hallo {{ first_name }}, dir werden {{ share_count }} Anteile zu insgesamt {{ payout_amount }} EUR ausbezahlt.",
            "template_id": tpl_id.to_string(),
            "repayment_phase_id": phase_id.to_string(),
        }))
        .send().await;
    assert_eq!(res.status(), 200, "Bulk-send must accept template_id + repayment_phase_id");

    // Wait for worker to process recipients (3 recipients * 36s send-interval would be too slow;
    // override interval to 0 in test setup OR poll for completion with timeout)
    test_server.wait_for_mail_worker_idle(std::time::Duration::from_secs(15)).await;

    // Assert: 3 MemberDocuments persisted, all status='sent' (or all 'failed' in stub-mode)
    let docs = test_server
        .query_documents_for_phase(phase_id, "repayment_mail")
        .await;
    assert_eq!(docs.len(), 3, "Expected 3 MemberDocuments, got {}: {:?}", docs.len(), docs);
    for d in &docs {
        assert!(d.template_id.is_some(), "All docs should have template_id (D-12)");
        assert!(d.mail_recipient_id.is_some(), "All docs should have mail_recipient_id");
        assert!(d.status.is_some(), "All docs should have status set");
        // In real-SMTP-mode: status="sent"; in stub-fail-mode: status="failed"
        // Either way, all 3 must be persisted -> N=3 documents = SC#3
    }
    assert_eq!(
        docs.iter().filter(|d| d.template_id == Some(tpl_id)).count(),
        3,
        "All 3 docs must reference the same template_id (MAIL-03)"
    );
}
